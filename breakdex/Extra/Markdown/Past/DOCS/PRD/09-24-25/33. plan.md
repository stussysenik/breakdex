        logger.info("🎬 VIDEO_MODEL: Async seek starting with robust readiness check.", metadata: ["target_time": "\(time.seconds)"])

        guard let player = self.avPlayer, let item = player.currentItem else {
            throw NSError(domain: "SeekFailed", code: 1, userInfo: [NSLocalizedDescriptionKey: "Player or player item is nil."])

        // 🎯 CRITICAL FIX: Instead of a brittle guard check, we now explicitly AWAIT readiness.
