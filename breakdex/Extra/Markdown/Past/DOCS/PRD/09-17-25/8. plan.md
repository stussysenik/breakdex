  Task {
      do {
          // Create output URL

              .appendingPathExtension("mov")

          // Determine trim range and rotation
          let startTime = trimStartTime ?? 0
          let endTime = trimEndTime ?? assetToSave.duration.seconds
          let timeRange = CMTimeRange(
              start: CMTime(seconds: startTime, preferredTimescale: 600),
              duration: CMTime(seconds: endTime - startTime, preferredTimescale:
  600)
          )

          // Export video with PROPER rotation applied
          let exportedURL = try await VideoTransformBuilder.exportVideo(
              asset: originalAsset, // Use original for full processing
              trimRange: timeRange,
              quarterTurns: rotationQuarterTurns,
              outputURL: outputURL
          )

          // Save to BreakDex album
          let photosIdentifier = try await
  BreakDexAlbumManager.shared.saveVideoToBreakDexAlbum(exportedURL)

          // Update Core Data
          let move = Move(context: viewContext)
          move.name = moveName
          move.videoURL = exportedURL
          move.photosIdentifier = photosIdentifier
          move.trimStartTime = startTime
          move.trimEndTime = endTime
          move.rotationQuarterTurns = Int16(rotationQuarterTurns)
          move.createdAt = Date()

          try viewContext.save()
      }
  }
