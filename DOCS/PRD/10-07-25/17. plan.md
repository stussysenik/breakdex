1.  **State Lifecycle Failure (The Core Issue):** The logs show multiple initializations of `AddMoveStateOwner`, each creating a new `AddMoveUnifiedState` instance (e.g., `ObjectIdentifier(0x...106c21800)`, `ObjectIdentifier(0x...10fee8400)`). 2.  **Race Condition (The Symptom):** The original instance continues its asynchronous loading task. The new instance, which the UI is now bound to, receives this progress update, determines it's in an invalid state, and transitions to `.error`. In categorical terms, the functor mapping your application state to your UI state is broken. Instead of preserving the single `AddMoveUnifiedState` object across view updates, it's creating new ones, breaking the structural isomorphism between your model and view.

// In: AddMoveUnifiedState.swift

@MainActor
public func didSelectVideo(_ item: PhotosPickerItem) {
    let identifier = item.itemIdentifier ?? "unknown"
    let correlationId = UUID().uuidString.prefix(8)

    logger.info("🎯 UNIFIED_LOADING_ENTRY: 🚀 VIDEO_SELECTION_START [\(correlationId)] - Processing video selection: \(identifier)")

    // 1. Cancel any previous loading task to prevent overlapping operations.
    videoLoadingTask?.cancel()

    // 2. Create a new, single Task to encapsulate the entire atomic operation.
    videoLoadingTask = Task { [weak self] in
        guard let self = self else { return }

        do {
            // STEP 1: Atomically reset the entire state to a clean slate.
            // This ensures no stale data from previous operations interferes.
            try await self.prepareForNewVideoSelection()
            try Task.checkCancellation()

            // STEP 2: SYNCHRONOUSLY transition to the .loadingVideo state.
            // This is the CRITICAL fix for the race condition. It guarantees the state is correct
            // before any async file loading begins.
            let transitionSuccess = self.beginLoadingState(from: self.flowState, videoIdentifier: identifier)
            guard transitionSuccess else {
                self.logger.error("❌ Initial state transition to .loadingVideo failed. Aborting.")
                await self.setError(message: "Failed to initialize loading state.")
                return
            }
            try Task.checkCancellation()

            // STEP 3: With the state now correctly set, perform asynchronous work.
            // Fetch file size for the loading UI.
            await self.fetchAndDisplayFileSize(from: item, correlationId: correlationId)
            try Task.checkCancellation()

            // Perform the actual, long-running video load.
            await self.loadVideoWithCancellation(from: item, correlationId: correlationId)

        } catch is CancellationError {
            logger.info("🛑 Video loading task was cancelled. [\(correlationId)]")
            await self.performLoadingCancellationCleanup(correlationId: correlationId)
        } catch {
            logger.error("❌ An unexpected error occurred during the video selection process: \(error.localizedDescription) [\(correlationId)]")
            await self.setError(message: "A problem occurred while selecting the video.", underlying: error.localizedDescription)
        }
    }
}
