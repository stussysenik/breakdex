**3. Implement the Interactive Timeline Components**

This feature requires two new reusable views. Create a new SwiftUI View file named `ComboTimelineView.swift`. This will be the horizontally scrolling container.

```swift
// In ComboTimelineView.swift
import SwiftUI

struct ComboTimelineView: View {
    @Binding var moves: [Move]
    @Binding var activeIndex: Int?

    var body: some View {
        ScrollViewReader { proxy in
            ScrollView(.horizontal, showsIndicators: false) {
                LazyHStack(spacing: 0) {
                    ForEach(Array(moves.enumerated()), id: \.element.id) { index, move in
                        HStack(spacing: 0) {
                            TimelineNodeView(
                                move: move,
                                sequenceNumber: index + 1,
                                isActive: activeIndex == index,
                                onDelete: {
                                    moves.remove(at: index)
                                    // Adjust active index if needed
                                    if activeIndex == index {
                                        activeIndex = moves.isEmpty ? nil : 0
                                    }
                                }
                            )
                            .onTapGesture {
                                activeIndex = index
                            }

                            // Add a connecting line if it's not the last node
                            if index < moves.count - 1 {
                                Rectangle()
                                    .frame(width: 30, height: 2)
                                    .foregroundColor(.gray)
                            }
                        }
                    }
                }
                .padding(.horizontal)
            }
            .onChange(of: activeIndex) { oldIndex, newIndex in
                if let newIndex {
                    withAnimation {
                        proxy.scrollTo(moves[newIndex].id, anchor: .center)
                    }
                }
            }
        }
    }
}
```

```swift
// In TimelineNodeView.swift
import SwiftUI

struct TimelineNodeView: View {
    let move: Move
    let sequenceNumber: Int
    let isActive: Bool
    let onDelete: () -> Void

    var body: some View {
        ZStack(alignment: .topTrailing) {
            // The main node circle
            Circle()
                .strokeBorder(isActive ? Color.accent : Color.stateNew, lineWidth: isActive ? 4 : 2)
                .background(Circle().fill(Color.neutralFill))
                .frame(width: isActive ? 66 : 50, height: isActive ? 66 : 50)
                .overlay(
                    Text("\(sequenceNumber)")
                        .font(isActive ? .title2.bold() : .body)
                        .foregroundColor(.textPrimary)
                )

            // The delete button, only shown if the node is active
            if isActive {
                Button(action: onDelete) {
                    Image(systemName: "xmark.circle.fill")
                        .font(.title2)
                        .foregroundColor(.red)
                        .background(Circle().fill(Color.white))
                }
                .offset(x: 8, y: -8)
                .transition(.scale.combined(with: .opacity))
            }
        }
        .padding(.horizontal, 15) // Adds space between nodes
        .animation(.spring(response: 0.4, dampingFraction: 0.6), value: isActive)
    }
}
```

```swift
private func saveCombo() {
    let newCombo = Combo(context: viewContext)
    newCombo.id = UUID()
    newCombo.name = "New Combo \(Date().formatted())" // Placeholder name

    for (index, move) in comboMoves.enumerated() {
        let comboMove = ComboMove(context: viewContext)
        comboMove.id = UUID()
        comboMove.sequenceIndex = Int64(index)
        comboMove.move = move
        comboMove.combo = newCombo
    }

    do {
        try viewContext.save()
```
