```swift
private func makeDragGesture(isStartHandle: Bool, totalWidth: CGFloat) -> some Gesture {
    DragGesture()
        .onChanged { dragValue in
            if !viewModel.isDraggingHandle {
                // --- On drag start ---
                withAnimation(.spring(response: 0.3, dampingFraction: 0.6)) {
                    viewModel.isDraggingHandle = true
                }
                viewModel.triggerHapticFeedback(for: .dragStart)
            }
            
            // MARK: - New Clamping Logic
            
            if isStartHandle {
                // 1. Calculate the proposed new pixel position for the start handle
                let newPosition = startHandlePosition(in: totalWidth) + dragValue.translation.width
                
                // 2. Define the boundaries: 0 to (end handle position - minimum gap)
                let minimumPixelGap = totalWidth * (viewModel.oneFrameDuration / viewModel.videoDuration)
                let maxPosition = endHandlePosition(in: totalWidth) - minimumPixelGap
                
                // 3. Clamp the position within the valid boundaries
                let clampedPosition = max(0, min(newPosition, maxPosition))
                
                // 4. Convert the safe, clamped position back to a time and snap it
                let proposedTime = positionToTime(clampedPosition, totalWidth: totalWidth)
                let snappedTime = round(proposedTime / viewModel.oneFrameDuration) * viewModel.oneFrameDuration
                
                // 5. Update the view model and trigger haptics if the frame changes
                if abs(snappedTime - viewModel.startTime) > viewModel.oneFrameDuration / 2 {
                    selectionFeedback.selectionChanged()
                }
                viewModel.startTime = snappedTime
                
            } else { // This is the End Handle
                // 1. Calculate the proposed new pixel position for the end handle
                let newPosition = endHandlePosition(in: totalWidth) + dragValue.translation.width
                
                // 2. Define the boundaries: (start handle position + minimum gap) to the total width
                let minimumPixelGap = totalWidth * (viewModel.oneFrameDuration / viewModel.videoDuration)
                let minPosition = startHandlePosition(in: totalWidth) + minimumPixelGap

                // 3. Clamp the position within the valid boundaries
                let clampedPosition = min(totalWidth, max(newPosition, minPosition))
                
                // 4. Convert the safe, clamped position back to a time and snap it
                let proposedTime = positionToTime(clampedPosition, totalWidth: totalWidth)
                let snappedTime = round(proposedTime / viewModel.oneFrameDuration) * viewModel.oneFrameDuration
                
                // 5. Update the view model and trigger haptics if the frame changes
                if abs(snappedTime - viewModel.endTime) > viewModel.oneFrameDuration / 2 {
                    selectionFeedback.selectionChanged()
                }
                viewModel.endTime = snappedTime
            }
        }
        .onEnded { _ in
            // --- On drag end ---
            withAnimation(.spring(response: 0.3, dampingFraction: 0.6)) {
                viewModel.isDraggingHandle = false
            }
            viewModel.triggerHapticFeedback(for: .dragEnd)
        }
}

```

### Summary of the Fix

  * **Position-First Clamping**: Instead of calculating a new time and then clamping it, we now calculate the handle's new **pixel position** first.
  * **Clear Boundaries**: We define a valid pixel range for each handle on every drag update. The start handle cannot go past the end handle (minus a small gap), and the end handle cannot go before the start handle.
  * **Reliable Constraints**: By clamping the position *before* converting it to a time, the handles will physically stop at the boundaries, preventing them from going out of bounds or crossing over, resulting in a much more stable and predictable slider.
