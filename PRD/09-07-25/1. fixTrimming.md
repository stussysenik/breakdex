// TrimControlPanelView.swift (Corrected)
import SwiftUI
import AVKit

struct TrimControlPanelView: View {
    @ObservedObject var viewModel: TrimmerViewModel

    // MARK: - State Properties
    @State private var currentZoom: CGFloat = 1.0
    @GestureState private var isMagnifying: Bool = false
    
    @State private var initialStartTime: CMTime = .zero
    @State private var initialEndTime: CMTime = .zero
    
    private let handleWidth: CGFloat = 20
    private let selectionFeedback = UISelectionFeedbackGenerator()
    
    // MARK: - Body
    var body: some View {
        VStack(spacing: 0) {
            if viewModel.videoDuration.seconds > 0 {
                trimmerTimelineView
                    .padding(.top, 12)
            } else {
                placeholderTimelineView
            }
            timeDisplayView
                .padding(.top, 8)
        }
        .padding(.horizontal, 24)
        .padding(.top)
    }
    
    // MARK: - UI Components
    private var trimmerTimelineView: some View {
        GeometryReader { geometry in
            let totalWidth = geometry.size.width
            
            ZStack(alignment: .leading) {
                // Background timeline bar
                Rectangle().fill(Color.secondary.opacity(0.5))
                    .frame(height: 6)
                    .cornerRadius(3)
                    .allowsHitTesting(false)
                
                // Active range
                Rectangle().fill(Color.accentColor)
                    .frame(width: activeRangeWidth(in: totalWidth), height: 6)
                    .cornerRadius(3)
                    .offset(x: activeRangeOffset(in: totalWidth))
                    .allowsHitTesting(false)
                
                // Start Handle
                handleView()
                    .offset(x: startHandlePosition(in: totalWidth) - (handleWidth / 2))
                    .overlay(
                        PanGestureRecognizerRepresentable(
                            onChanged: { gesture in
                                self.handlePanChanged(gesture: gesture, isStartHandle: true, totalWidth: totalWidth)
                            },
                            onEnded: { gesture in
                                self.handlePanEnded(gesture: gesture)
                            }
                        )
                    )
                
                // End Handle
                handleView()
                    .offset(x: endHandlePosition(in: totalWidth) - (handleWidth / 2))
                    .overlay(
                        PanGestureRecognizerRepresentable(
                            onChanged: { gesture in
                                self.handlePanChanged(gesture: gesture, isStartHandle: false, totalWidth: totalWidth)
                            },
                            onEnded: { gesture in
                                self.handlePanEnded(gesture: gesture)
                            }
                        )
                    )
            }
        }
        .frame(height: 80)
        .accessibilityIdentifier("CustomRangeSlider")
    }
    
    private var placeholderTimelineView: some View {
        Rectangle()
            .fill(Color.gray.opacity(0.5))
            .frame(height: 60)
            .cornerRadius(12)
            .padding(.vertical, 16)
            .overlay(ProgressView())
    }
    
    private var timeDisplayView: some View {
        HStack {
            VStack(alignment: .leading) {
                Text("Start")
                    .font(.caption)
                    .foregroundColor(.secondary)
                Text(formatTime(viewModel.startTime))
                    .font(.system(size: 16, weight: .bold, design: .monospaced))
                    .foregroundColor(.accentColor)
            }
            Spacer()
            VStack {
                Text("Duration")
                    .font(.caption)
                    .foregroundColor(.secondary)
                Text(formatTime(viewModel.endTime - viewModel.startTime))
                    .font(.system(size: 14, weight: .regular, design: .monospaced))
                    .foregroundColor(.secondary)
            }
            Spacer()
            VStack(alignment: .trailing) {
                Text("End")
                    .font(.caption)
                    .foregroundColor(.secondary)
                Text(formatTime(viewModel.endTime))
                    .font(.system(size: 16, weight: .bold, design: .monospaced))
                    .foregroundColor(.accentColor)
            }
        }
        .padding(.top, 8)
    }
    
    @ViewBuilder
    private func handleView() -> some View {
        Circle()
            .fill(Color.accentColor)
            .frame(width: handleWidth, height: handleWidth)
            .scaleEffect(viewModel.isDraggingHandle ? 1.2 : 1.0)
            .overlay(Circle().stroke(Color.white, lineWidth: 2))
            .frame(width: 44, height: 44) // Hit target
            .contentShape(Rectangle())

    // MARK: - Gestures
    private func handlePanBegan() {
        initialStartTime = viewModel.startTime
        initialEndTime = viewModel.endTime
        withAnimation(.spring(response: 0.3, dampingFraction: 0.6)) {
            viewModel.isDraggingHandle = true
        }
        viewModel.triggerHapticFeedback(for: .dragStart)
    }

    private func handlePanChanged(gesture: UIPanGestureRecognizer, isStartHandle: Bool, totalWidth: CGFloat) {
        if !viewModel.isDraggingHandle { handlePanBegan() }

        var transaction = Transaction()
        transaction.disablesAnimations = true

        withTransaction(transaction) {
            let translationX = gesture.translation(in: gesture.view).x
            let verticalDrag = abs(gesture.translation(in: gesture.view).y)
            let precisionScale = 1.0 - (min(1.0, max(0, verticalDrag / 100.0))) * 0.9
            
            if isStartHandle {
                let initialPosition = timeToPosition(initialStartTime, totalWidth: totalWidth)
                let newPosition = initialPosition + (translationX * precisionScale)
                let proposedTime = positionToTime(newPosition, totalWidth: totalWidth)
                let maxTime = initialEndTime - viewModel.oneFrameDuration
                let clampedTime = Quantizer.quantizedTime(for: proposedTime, in: viewModel.videoDuration, frameDuration: viewModel.oneFrameDuration, minTime: .zero, maxTime: maxTime)
                
                if didFrameChange(from: viewModel.startTime, to: clampedTime) { selectionFeedback.selectionChanged() }
                viewModel.startTime = clampedTime
                viewModel.requestSeek(to: clampedTime)
            } else { // End Handle
                let initialPosition = timeToPosition(initialEndTime, totalWidth: totalWidth)
                let newPosition = initialPosition + (translationX * precisionScale)
                let proposedTime = positionToTime(newPosition, totalWidth: totalWidth)
                let minTime = initialStartTime + viewModel.oneFrameDuration
                let clampedTime = Quantizer.quantizedTime(for: proposedTime, in: viewModel.videoDuration, frameDuration: viewModel.oneFrameDuration, minTime: minTime, maxTime: viewModel.videoDuration)

                if didFrameChange(from: viewModel.endTime, to: clampedTime) { selectionFeedback.selectionChanged() }
                viewModel.endTime = clampedTime
                viewModel.requestSeek(to: clampedTime)
            }
        }
    }

    private func handlePanEnded(gesture: UIPanGestureRecognizer) {
        withAnimation(.spring(response: 0.3, dampingFraction: 0.6)) {
            viewModel.isDraggingHandle = false
        }
        viewModel.triggerHapticFeedback(for: .dragEnd)
    }

    private func didFrameChange(from oldTime: CMTime, to newTime: CMTime) -> Bool {
        let oldFrame = round(oldTime.seconds / viewModel.oneFrameDuration.seconds)
        let newFrame = round(newTime.seconds / viewModel.oneFrameDuration.seconds)
        return oldFrame != newFrame
    }
    
    // MARK: - Helper Methods (Corrected)
    private func startHandlePosition(in totalWidth: CGFloat) -> CGFloat {
        return timeToPosition(viewModel.startTime, totalWidth: totalWidth)
    }
    
    private func endHandlePosition(in totalWidth: CGFloat) -> CGFloat {
        return timeToPosition(viewModel.endTime, totalWidth: totalWidth)
    }

    private func activeRangeOffset(in totalWidth: CGFloat) -> CGFloat {
        return startHandlePosition(in: totalWidth)
    }
    
    private func activeRangeWidth(in totalWidth: CGFloat) -> CGFloat {
        return endHandlePosition(in: totalWidth) - startHandlePosition(in: totalWidth)
    }

    private func timeToPosition(_ time: CMTime, totalWidth: CGFloat) -> CGFloat {
        guard viewModel.videoDuration.seconds > 0 else { return 0 }
        let percentage = time.seconds / viewModel.videoDuration.seconds
        
        // This is the usable track width for the handles' centers
        let usableWidth = totalWidth - handleWidth
        return (percentage * usableWidth) + (handleWidth / 2)
    }
    
    private func positionToTime(_ position: CGFloat, totalWidth: CGFloat) -> CMTime {
        guard totalWidth > 0 else { return .zero }
        let usableWidth = totalWidth - handleWidth
        let percentage = (position - (handleWidth / 2)) / usableWidth
        let seconds = percentage * viewModel.videoDuration.seconds
        return CMTime(seconds: seconds, preferredTimescale: viewModel.videoDuration.timescale)
    }
    
    private func formatTime(_ time: CMTime) -> String {
        let seconds = CMTimeGetSeconds(time)
        guard seconds.isFinite else { return "--:--.--" }
        let minutes = Int(seconds) / 60
        let remainingSeconds = Int(seconds) % 60
        let milliseconds = Int((seconds.truncatingRemainder(dividingBy: 1)) * 100)
        return String(format: "%02d:%02d.%02d", minutes, remainingSeconds, milliseconds)
    }
}
