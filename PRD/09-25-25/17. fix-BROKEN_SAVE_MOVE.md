This non-commutative diagram—where `cleanup` followed by `transition` does not equal `transition` followed by `cleanup`—causes the `TrimmerViewModel` to leak, holding onto memory and potentially causing crashes or undefined behavior when the user tries to add another move.

import AVFoundation
import UIKit

enum ExportError: Error {
    case exportFailed(Error?)
    case unsupportedFileType
}

class VideoExporter {
    
    /// Exports an AVAsset with specified trim range and rotation.
    /// This is the corrected 'applyTrimAndRotate' morphism.
    static func export(asset: AVAsset, trimStartTime: Double, trimEndTime: Double, rotation: Int) async throws -> AVAsset {
        guard let exportSession = AVAssetExportSession(asset: asset, presetName: AVAssetExportPresetHighestQuality) else {
            throw ExportError.exportFailed(nil)
        }

        let timeRange = CMTimeRange(
            start: CMTime(seconds: trimStartTime, preferredTimescale: asset.duration.timescale),
            end: CMTime(seconds: trimEndTime, preferredTimescale: asset.duration.timescale)
        )
        
        let outputURL = FileManager.default.temporaryDirectory
            .appendingPathComponent(UUID().uuidString)
            .appendingPathExtension("mov")

        exportSession.outputURL = outputURL
        exportSession.outputFileType = .mov
        exportSession.timeRange = timeRange
        
        // Only apply video composition if rotation is needed.
        if rotation != 0, let videoTrack = asset.tracks(withMediaType: .video).first {
            exportSession.videoComposition = createVideoComposition(for: videoTrack, rotation: rotation)
        }

        await exportSession.export()

        if let error = exportSession.error {
            throw ExportError.exportFailed(error)
        }

        return AVURLAsset(url: outputURL)
    }

    private static func createVideoComposition(for track: AVAssetTrack, rotation: Int) -> AVMutableVideoComposition {
        let videoComposition = AVMutableVideoComposition()
        let instruction = AVMutableVideoCompositionInstruction()
        let layerInstruction = AVMutableVideoCompositionLayerInstruction(assetTrack: track)
        
        let transform = calculateTransform(for: track, rotation: rotation)
        layerInstruction.setTransform(transform, at: .zero)

        // Correctly set render size by swapping dimensions for 90/270 degree rotations
        var naturalSize = track.naturalSize
        if rotation == 1 || rotation == 3 { // 90 or 270 degrees
            swap(&naturalSize.width, &naturalSize.height)
        }
        videoComposition.renderSize = naturalSize
        
        instruction.timeRange = track.timeRange
        instruction.layerInstructions = [layerInstruction]
        videoComposition.instructions = [instruction]
        videoComposition.frameDuration = CMTime(value: 1, timescale: 30) // Standard frame rate

        return videoComposition
    }

    private static func calculateTransform(for track: AVAssetTrack, rotation: Int) -> CGAffineTransform {
        let size = track.naturalSize
        var transform = track.preferredTransform

        switch rotation {
        case 1: // 90 degrees clockwise
            transform = transform.concatenating(CGAffineTransform(rotationAngle: .pi / 2))
            transform = transform.concatenating(CGAffineTransform(translationX: size.height, y: 0))
        case 2: // 180 degrees
            transform = transform.concatenating(CGAffineTransform(rotationAngle: .pi))
            transform = transform.concatenating(CGAffineTransform(translationX: size.width, y: size.height))
        case 3: // 270 degrees clockwise
            transform = transform.concatenating(CGAffineTransform(rotationAngle: .pi * 1.5))
            transform = transform.concatenating(CGAffineTransform(translationX: 0, y: size.width))
        default:
            break
        }
        
        return transform
    }
}
