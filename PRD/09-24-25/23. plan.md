//  AVPlayerViewRepresentable.swift
//  BreakingFlashcards
//
//  Single Responsibility: Provide reliable AVPlayer rendering in SwiftUI
//  Replaces SwiftUI VideoPlayer with AVFoundation direct access for stability
//  UPDATED: Implemented Coordinator pattern to handle race conditions where the
//  player's currentItem is not yet ready when the view is created.
//

import SwiftUI
import AVFoundation
import OSLog

struct AVPlayerViewRepresentable: UIViewRepresentable {
    private let diagnosticLogger = DiagnosticLoggingHelper(category: "AVPlayerViewRepresentable")
    private let logger = Logger(subsystem: "com.breakingflashcards", category: "AVPlayerView")

    let viewModel: any VideoPlayerViewModelProtocol

    init(viewModel: any VideoPlayerViewModelProtocol) {
        self.viewModel = viewModel
    }

    // MARK: - Coordinator
    
    func makeCoordinator() -> Coordinator {
        Coordinator(viewModel: viewModel, logger: logger, diagnosticLogger: diagnosticLogger)
    }
    
    class Coordinator: NSObject {
        let viewModel: any VideoPlayerViewModelProtocol
        let logger: Logger
        let diagnosticLogger: DiagnosticLoggingHelper
        var playerItemStatusObserver: NSKeyValueObservation?

        init(viewModel: any VideoPlayerViewModelProtocol, logger: Logger, diagnosticLogger: DiagnosticLoggingHelper) {
            self.viewModel = viewModel
            self.logger = logger
            self.diagnosticLogger = diagnosticLogger
            super.init()
        }

        func setupObservers(for playerView: PlayerView) {
            playerItemStatusObserver?.invalidate()
            playerItemStatusObserver = viewModel.avPlayer?.currentItem?.observe(\.status, options: [.new, .initial]) { [weak playerView] item, _ in
                DispatchQueue.main.async {
                    if item.status == .readyToPlay {
                        // This ensures the layer is updated once the item is ready.
                        if playerView?.playerLayer.player !== self.viewModel.avPlayer {
                           playerView?.playerLayer.player = self.viewModel.avPlayer
                        }
                        self.logger.info("🎬 COORD: Player item is ready. Layer updated.")
                    }
                }
            }
        }
        
        deinit {
            playerItemStatusObserver?.invalidate()
        }
    }

    // MARK: - UIViewRepresentable Methods

    func makeUIView(context: Context) -> PlayerView {
        diagnosticLogger.startTiming("make_uiview")
        logger.info("🎬 AV_PLAYER_VIEW: makeUIView called")

        let playerView = PlayerView()
        playerView.playerLayer.videoGravity = .resizeAspect
        
        // Let the coordinator handle assigning the player when the item is ready
        context.coordinator.setupObservers(for: playerView)
        
        if viewModel.avPlayer?.currentItem?.status == .readyToPlay {
            playerView.playerLayer.player = viewModel.avPlayer
        }

        let tapGesture = UITapGestureRecognizer(target: playerView, action: #selector(PlayerView.handleTap))
        playerView.addGestureRecognizer(tapGesture)
        playerView.isUserInteractionEnabled = true
        
        logger.info("🎬 AV_PLAYER_VIEW: PlayerView created successfully with gesture recognizer.")
        diagnosticLogger.stopTiming("make_uiview")
        return playerView
    }
    
    func updateUIView(_ uiView: PlayerView, context: Context) {
        logger.info("🎬 AV_PLAYER_VIEW: updateUIView called")
        
        // Ensure the coordinator is observing the correct item
        if uiView.playerLayer.player !== viewModel.avPlayer {
             context.coordinator.setupObservers(for: uiView)
        }
        
        // If the item is already ready, ensure the player is set.
        if viewModel.avPlayer?.currentItem?.status == .readyToPlay {
            if uiView.playerLayer.player !== viewModel.avPlayer {
                uiView.playerLayer.player = viewModel.avPlayer
                logger.info("🎬 AV_PLAYER_VIEW: Player updated in updateUIView because item was ready.")
            }
        }
    }
    
    static func dismantleUIView(_ uiView: PlayerView, coordinator: Coordinator) {
        uiView.playerLayer.player = nil
        coordinator.playerItemStatusObserver?.invalidate()
    }

    // MARK: - PlayerView (UIView Subclass)
    
    class PlayerView: UIView {
        private let logger = Logger(subsystem: "com.breakingflashcards", category: "PlayerView")

        override static var layerClass: AnyClass { AVPlayerLayer.self }
        var playerLayer: AVPlayerLayer { layer as! AVPlayerLayer }

        override init(frame: CGRect) {
            super.init(frame: frame)
            playerLayer.backgroundColor = UIColor.black.cgColor
            logger.info("🎬 PLAYER_VIEW: init called")
        }

        required init?(coder: NSCoder) {
            fatalError("init(coder:) has not been implemented")
        }

        override func layoutSubviews() {
            super.layoutSubviews()
            logger.info("🎬 PLAYER_VIEW: layoutSubviews called - Frame: \(self.frame)")
        }

        @objc func handleTap() {
            guard let player = playerLayer.player else {
                logger.warning("🎬 PLAYER_VIEW: Tap received but no player assigned")
                return
            }
            logger.info("🎬 PLAYER_VIEW: Video tapped - toggling playback. Current rate: \(player.rate)")
            if player.rate == 0 {
                player.play()
            } else {
                player.pause()
            }
        }
        
        deinit {
            logger.info("🗑️ PlayerView deinitialized")
        }
    }
}
