  // Enhanced coalescing for buttery-smooth scrubbing
  private func setupEnhancedScrubbing() {
      // High-frequency display link for 60fps scrubbing
      displayLink = CADisplayLink(

  }

  @objc private func preciseTick() {
      guard let pendingTime = pendingPreviewTime else { return }

      // Frame-accurate seeking with zero tolerance
      player.seek(
          to: pendingTime,
          toleranceBefore: .zero,
          toleranceAfter: .zero
      ) { [weak self] _ in
          self?.handleSeekCompletion()
      }
  }

  Advanced Haptic System:

  // Multi-level haptic feedback
  enum HapticPattern {
      case dragStart      // Medium impact
      case dragEnd        // Medium impact  
      case frameSnap      // Light selection
      case minimumBump    // Heavy impact (physical barrier)
      case rotationClick  // Light selection
  }

  private func triggerHaptic(_ pattern: HapticPattern) {
      switch pattern {
      case .dragStart, .dragEnd:
          UIImpactFeedbackGenerator(style: .medium).impactOccurred()
      case .frameSnap:
          UISelectionFeedbackGenerator().selectionChanged()
      case .minimumBump:
          UIImpactFeedbackGenerator(style: .heavy).impactOccurred()
      case .rotationClick:
          UISelectionFeedbackGenerator().selectionChanged()
      }
  }
