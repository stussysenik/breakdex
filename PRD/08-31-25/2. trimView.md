## 4. Precise trimming component  

We already have `PreciseVideoTrimmerView.swift` and `VideoTrimmerView.swift`. Keep one:

1. **Wrap in a coordinator for AVAssetExportSession progress.**  
2. **Expose two-way binding** `@Binding var trimmedURL: URL?`.

```swift
struct TrimVideoView: View {
    let asset: AVAsset
    let originalURL: URL
    @State private var trimmedURL: URL?
    @State private var isExporting = false
    @Binding var state: AddMoveState

    var body: some View {
        VStack(spacing: 8) {
            PreciseVideoTrimmerView(asset: asset, onExport: export)
            if isExporting {
                ProgressView("Exporting…", value: exportProgress,
                             total: 1).progressViewStyle(.linear)
            }
            HStack {
                Button("Back") { state = .previewing(asset: asset, url: originalURL) }
                Button("Save Trim") {
                    guard let url = trimmedURL else { return }
                    state = .naming(url: url)
                }
                .disabled(trimmedURL == nil)
            }
            .buttonStyle(.borderedProminent)
        }
        .navigationTitle("Trim")
    }
}
```
`PreciseVideoTrimmerView` outputs `startTime`/`endTime`; the export helper:

```swift
func export(start: CMTime, end: CMTime) async throws -> URL
```

Runs in a Task, reports progress via `AVAssetExportSession.progress`.

## 6. Error surfacing  

Inside `AddMoveViewModel`, wrap every async call:

```swift
do { … } catch {
    logger.error("AddMove failure: \(error.localizedDescription)")
    await MainActor.run { self.state = .error(message: error.userFacingMessage) }
}
```

`userFacingMessage` is an extension translating common errors:
```swift
enum AddMoveError: Error {
    case icloudUnavailable, fileMoveFailed, exportCancelled, unknown
}

extension Error {
    var userFacingMessage: String {
        switch self as? AddMoveError {
        case .icloudUnavailable: return "iCloud not accessible."
        case .fileMoveFailed:    return "Couldn’t copy video locally."
        case .exportCancelled:   return "Export was cancelled."
        default:                 return "Something went wrong."
        }
    }
}
```

Present with `.alert`.

## 8. Glue code – the new container  

```swift
struct AddMoveContainer: View {
    @StateObject private var vm = AddMoveViewModel()

    var body: some View {
        switch vm.state {
        case .ready:
            AddMoveSelectClipView().environmentObject(vm)
        case .loading(let p, let status):
            LoadingView(progress: p, status: status)
        case .previewing(let asset, let url):
            VideoPreviewOptionsView(asset: asset, url: url, state: $vm.state)
        case .trimming(let asset, let url):
            TrimVideoView(asset: asset, originalURL: url, state: $vm.state)
        case .naming(let url):
            NameMoveView(videoURL: url, onSave: vm.handleSave)
        case .saving:
            SavingView()
        case .success(let msg):
            MoveAddedSuccessView(message: msg, onDone: { vm.reset() })
        case .error(let msg):
            ErrorView(message: msg, onRetry: { vm.reset() })
        }
    }
}
```
Embed `AddMoveContainer` in the existing **Add** tab.
