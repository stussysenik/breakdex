## Diagnosis: A Silent Mutability Error

The logs show that our `VideoTransformBuilder` is now calculating a **perfectly correct** final transform matrix, including both rotation and translation. This is a huge success.

However, the preview still hangs. Why?

The root cause is that `AVVideoCompositionInstruction`'s `layerInstructions` property, while belonging to a mutable object, contains an array of **immutable** `AVVideoCompositionLayerInstruction` objects. When we try to call `setTransform` on them, the change **fails silently**. The code runs without error, but the transform is never actually applied to the composition.

We are building a mathematically perfect transform but failing to attach it to the video. The result is a composition that is still invalid, causing the `AVPlayerItem` to hang.

```swift
static func build(asset: AVAsset, trimRange: CMTimeRange? = nil, quarterTurns: Int) async throws -> (composition: AVMutableComposition, videoComposition: AVMutableVideoComposition) {
    let composition = AVMutableComposition()

    // 1. Create the trimmed composition. Its `timeRange` is our source of truth.
    let timeRange: CMTimeRange
    if let trimRange = trimRange {
        timeRange = trimRange
    } else {
        let duration = try await asset.load(.duration)
        timeRange = CMTimeRange(start: .zero, duration: duration)
    }
    // Only proceed if the asset has tracks
    guard try await asset.load(.hasProtectedContent) == false,
          let assetVideoTrack = try await asset.loadTracks(withMediaType: .video).first else {
        // Handle cases where the asset is protected or has no video
        throw NSError(domain: "VideoTransformBuilder", code: 1, userInfo: [NSLocalizedDescriptionKey: "Asset is protected or has no video tracks."])
    }
    
    // Insert the trimmed time range of the asset's video and audio tracks into our new composition
    if let compositionVideoTrack = composition.addMutableTrack(withMediaType: .video, preferredTrackID: kCMPersistentTrackID_Invalid) {
        try compositionVideoTrack.insertTimeRange(timeRange, of: assetVideoTrack, at: .zero)
    }
    if let assetAudioTrack = try await asset.loadTracks(withMediaType: .audio).first,
       let compositionAudioTrack = composition.addMutableTrack(withMediaType: .audio, preferredTrackID: kCMPersistentTrackID_Invalid) {
        try compositionAudioTrack.insertTimeRange(timeRange, of: assetAudioTrack, at: .zero)
    }

    // 2. Let AVFoundation create a base video composition.
    let videoComposition = AVMutableVideoComposition(propertiesOf: composition)
    
    // 3. Synchronize the instruction's timeline with the trimmed composition's timeline.
    if let instruction = videoComposition.instructions.first as? AVMutableVideoCompositionInstruction {
        instruction.timeRange = timeRange
    }
    
    // 4. If no rotation is needed, we are done.
    if quarterTurns % 4 == 0 {
        return (composition, videoComposition)
    }
    
    // 5. If rotation is needed, modify the composition AVFoundation made for us.
    let originalSize = videoComposition.renderSize
    var newRenderSize = originalSize
    
    let userRotation = CGAffineTransform(rotationAngle: .pi / 2 * CGFloat(quarterTurns))
    var translation: CGAffineTransform
    
    switch quarterTurns {
    case 1: // 90°
        newRenderSize = CGSize(width: originalSize.height, height: originalSize.width)
        translation = CGAffineTransform(translationX: originalSize.height, y: 0)
    case 2: // 180°
        translation = CGAffineTransform(translationX: originalSize.width, y: originalSize.height)
    case 3: // 270°
        newRenderSize = CGSize(width: originalSize.height, height: originalSize.width)
        translation = CGAffineTransform(translationX: 0, y: originalSize.width)
    default:
        translation = .identity
    }
    videoComposition.renderSize = newRenderSize
    
    // 6. --- THE DEFINITIVE FIX: MUTABLE COPY ---
    // Modify a mutable copy of the layer instructions.
    if let instruction = videoComposition.instructions.first as? AVMutableVideoCompositionInstruction,
       let assetTrack = composition.tracks(withMediaType: .video).first {
        
        var modifiedLayerInstructions: [AVVideoCompositionLayerInstruction] = []
        for layerInstruction in instruction.layerInstructions {
            // Create a mutable copy to work with
            let mutableLayerInstruction = layerInstruction.mutableCopy() as! AVMutableVideoCompositionLayerInstruction
            
            var originalTransform: CGAffineTransform = .identity
            if mutableLayerInstruction.getTransformRamp(for: assetTrack, at: .zero, start: &originalTransform, end: nil) {
                // Apply our transforms to the copy
                let finalTransform = originalTransform.concatenating(userRotation).concatenating(translation)
                mutableLayerInstruction.setTransform(finalTransform, for: assetTrack)
            }
            modifiedLayerInstructions.append(mutableLayerInstruction)
        }
        // Replace the original instructions with our modified ones
        instruction.layerInstructions = modifiedLayerInstructions
    }
    
    return (composition, videoComposition)
}
```
