func proposeStartTime(_ proposedTime: CMTime) {
    let frameDuration = oneFrameDuration
    let minimumDuration = CMTime(seconds: 1.0, preferredTimescale: videoDuration.timescale)
    
    // Clamp to video boundaries first
    var clampedTime = max(.zero, min(proposedTime, videoDuration))
    
    // ENFORCE minimum duration (this is the guardrail)
    if (endTime - clampedTime) < minimumDuration {
