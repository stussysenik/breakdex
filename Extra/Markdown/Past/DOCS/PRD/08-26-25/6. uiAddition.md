    // Enum for type-safe access to font weights.
    // Replace with your font's PostScript names.

**Step-by-Step Implementation Guide:**

**1. Define the IBM Color Palette**

First, create a new Swift file named `Color+Extensions.swift`. Add the following code to make the IBM button colors available throughout the app. These specific hex values are chosen from the IBM Design Language documentation to represent error, warning, and success states.[1][2]

```swift
import SwiftUI

extension Color {
    // Define colors from the IBM Design Language for review buttons.
    // Red 60 for "AGAIN" (Error/Danger)
    static let ibmErrorRed = Color(hex: "da1e28")
    // Orange 40 for "HARD" (Warning)
    static let ibmWarningOrange = Color(hex: "ff832b")
    // Green 50 for "GOOD" (Success)
    static let ibmSuccessGreen = Color(hex: "24a148")
}

// Helper to allow initializing Color with a hex string.
extension Color {
    init(hex: String) {
        let scanner = Scanner(string: hex.trimmingCharacters(in: .alphanumerics.inverted))
        var hexNumber: UInt64 = 0
        scanner.scanHexInt64(&hexNumber)
        
        let r = Double((hexNumber & 0xff0000) >> 16) / 255
        let g = Double((hexNumber & 0x00ff00) >> 8) / 255
        let b = Double(hexNumber & 0x0000ff) / 255
        
        self.init(red: r, green: g, blue: b)
    }
}
```

import SwiftUI

struct FlashcardReviewView: View {
    let learningState: String
    
    // Fetch moves matching the specified learning state.
    @FetchRequest private var moves: FetchedResults<Move>
    
    init(learningState: String) {
        self.learningState = learningState
        self._moves = FetchRequest<Move>(
            sortDescriptors: [NSSortDescriptor(keyPath: \Move.createdAt, ascending: true)],
            predicate: NSPredicate(format: "learningState == %@", learningState)
        )
    }

    var body: some View {
        // We will implement a TabView or similar to swipe between flashcards.
        // For now, this just shows the first move.
        if let move = moves.first {
            VStack {
                // Display the move's name and date
                Text(move.name ?? "Unknown Move")
                    .font(.appFont(.bold, size: 34))
                Text(move.createdAt ?? Date(), style: .date)
                    .font(.appFont(.regular, size: 17))
                    .foregroundColor(.secondary)
                
                // Placeholder for the CustomVideoPlayerView or ComboTimelineView
                Rectangle()
                    .fill(Color.secondary)
                    .aspectRatio(16/9, contentMode: .fit)
                    .cornerRadius(12)
                    .padding()
                
                Spacer()
                
                // Review action buttons
                ReviewButtons()
            }
            .padding()
            .navigationTitle(learningState)
        } else {
            VStack {
                Spacer()
                Text("No moves to review in this category.")
                    .multilineTextAlignment(.center)
                Spacer()
            }
        }
    }
}
