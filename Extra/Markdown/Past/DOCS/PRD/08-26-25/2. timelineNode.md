import SwiftUI
import AVKit

struct CreateComboView: View {
    @Environment(\.managedObjectContext) private var viewContext
    @FetchRequest(sortDescriptors: [NSSortDescriptor(keyPath: \Move.createdAt, ascending: true)])
    private var allMoves: FetchedResults<Move>

    @State private var comboMoves: [Move] = []
    @State private var activeNodeIndex: Int? = nil
    @State private var isMovePickerPresented = false

    var body: some View {
        VStack(spacing: 0) {
            // Top Section: Add Move Button
            Button("+ Add Move to Combo") {
                isMovePickerPresented = true
            }
            .padding()

            // Middle Section: Video Player or Empty State
            if let activeMove = activeMove {
                VideoPlayer(player: AVPlayer(url: videoURL(for: activeMove)))
                    .frame(height: 300)
            } else {
                ContentUnavailableView("Your Combo Is Empty", systemImage: "wand.and.stars")
                    .frame(height: 300)
            }

            Spacer(minLength: 20)

            // Bottom Section: Timeline
            Text("Your Combo")
                .font(.headline)
            ComboTimelineView(moves: $comboMoves, activeIndex: $activeNodeIndex)
                .frame(height: 100)

            // Final Action: Save Button with corrected padding
            Button("Save Combo") { /* Save Logic */ }
                .buttonStyle(.borderedProminent)
                .disabled(comboMoves.isEmpty)
                .padding(.vertical, 20) // Corrected vertical padding
        }
        .sheet(isPresented: $isMovePickerPresented) {
            MovePickerSheet(allMoves: allMoves, selectedMoves: $comboMoves)
        }
    }
    
    private var activeMove: Move? {
        guard let activeNodeIndex, !comboMoves.isEmpty, comboMoves.indices.contains(activeNodeIndex) else {
            return nil
        }
        return comboMoves[activeNodeIndex]
    }

    private func videoURL(for move: Move) -> URL {
        // This assumes videoReference stores a string path. Adapt if it stores raw bookmark data.
        let path = String(data: move.videoReference ?? Data(), encoding: .utf8) ?? ""
        return URL(filePath: path)
    }
}

// In ComboTimelineView.swift
import SwiftUI

struct ComboTimelineView: View {
    @Binding var moves: [Move]
    @Binding var activeIndex: Int?

    var body: some View {
        ScrollViewReader { proxy in
            ScrollView(.horizontal, showsIndicators: false) {
                HStack(spacing: 0) { // Use regular HStack for this layout
                    ForEach(Array(moves.enumerated()), id: \.element.id) { index, move in
                        HStack(spacing: 0) {
                            TimelineNodeView(
                                sequenceNumber: index + 1,
                                isActive: activeIndex == index,
                                onDelete: {
                                    moves.remove(at: index)
                                    // Reset active index intelligently
                                    if moves.isEmpty {
                                        activeIndex = nil
                                    } else if index <= (activeIndex ?? 0) {
                                        activeIndex = 0
                                    }
                                }
                            )
                            .onTapGesture { activeIndex = index }

                            if index < moves.count - 1 {
                                Rectangle().frame(width: 30, height: 2).foregroundColor(.gray)
                            }
                        }
                    }
                }
                .padding(.horizontal)
            }
            .onChange(of: activeIndex) { oldIndex, newIndex in
                if let newIndex, moves.indices.contains(newIndex) {
                    withAnimation { proxy.scrollTo(moves[newIndex].id, anchor: .center) }
                }
            }
        }
    }
}

import SwiftUI

struct TimelineNodeView: View {
    let sequenceNumber: Int
    let isActive: Bool
    let onDelete: () -> Void

    var body: some View {
        ZStack(alignment: .topTrailing) {
            Circle()
                .strokeBorder(isActive ? Color.accent : Color.stateNew, lineWidth: isActive ? 4 : 2)
                .background(Circle().fill(Color.neutralFill))
                .frame(width: isActive ? 66 : 50, height: isActive ? 66 : 50)
                .overlay(Text("\(sequenceNumber)").font(isActive ? .title2.bold() : .body).foregroundColor(.textPrimary))

            if isActive {
                Button(action: onDelete) {
                    Image(systemName: "xmark.circle.fill")
                        .font(.title2).foregroundColor(.red)
                        .background(Circle().fill(Color.white))
                }
                .offset(x: 8, y: -8)
                .transition(.scale.combined(with: .opacity))
            }
        }
        .padding(.horizontal, 15)
        .animation(.spring(response: 0.4, dampingFraction: 0.6), value: isActive)
    }
}

import SwiftUI

struct MovePickerSheet: View {
    @Environment(\.dismiss) var dismiss
    let allMoves: FetchedResults<Move>
    @Binding var selectedMoves: [Move]

    var body: some View {
        NavigationView {
            List(allMoves) { move in
                Button(action: {
                    selectedMoves.append(move)
                    dismiss()
                }) {
                    Text(move.name ?? "Untitled Move")
                }
            }
            .navigationTitle("Add Move to Combo")
            .toolbar {
                ToolbarItem(placement: .cancellationAction) {
                    Button("Cancel") {
                        dismiss()
                    }
                }
            }
        }
    }
}
